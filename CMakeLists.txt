cmake_minimum_required(VERSION 3.16)
project(cfl3d_nozzle LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(NOZZLE_ENABLE_OPENMP "Enable OpenMP CPU loop annotations" ON)
option(NOZZLE_ENABLE_CUDA "Build the native CUDA explicit backend when available" ON)
option(NOZZLE_ENABLE_KOKKOS "Build the Kokkos explicit backend when available" ON)
set(NOZZLE_CUDA_ARCHITECTURES "86" CACHE STRING "CUDA architectures for nozzle_cuda")
set(NOZZLE_KOKKOS_ROOT "" CACHE PATH "Root of a locally extracted Kokkos package")

add_executable(nozzle_solver src/nozzle_solver.cpp)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS src/nozzle_solver.cpp)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/src/nozzle_solver.cpp" NOZZLE_SOURCE_SHA)
target_compile_definitions(nozzle_solver PRIVATE NOZZLE_SOURCE_SHA="${NOZZLE_SOURCE_SHA}")
add_executable(quasi1d_reference reference/quasi1d.cpp)
add_executable(backend_geometry_test tests/backend_geometry_test.cpp)
add_executable(muscl_numerics_test tests/muscl_numerics_test.cpp)
target_compile_options(muscl_numerics_test PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(backend_geometry_test PRIVATE src)
target_compile_options(nozzle_solver PRIVATE -Wall -Wextra -Wpedantic)
target_compile_options(backend_geometry_test PRIVATE -Wall -Wextra -Wpedantic)
if(NOZZLE_ENABLE_OPENMP)
  find_package(OpenMP QUIET)
  if(OpenMP_CXX_FOUND)
    target_link_libraries(nozzle_solver PRIVATE OpenMP::OpenMP_CXX)
    target_compile_definitions(nozzle_solver PRIVATE NOZZLE_OPENMP=1)
  endif()
endif()

if(NOZZLE_ENABLE_CUDA)
  include(CheckLanguage)
  check_language(CUDA)
  if(CMAKE_CUDA_COMPILER)
    enable_language(CUDA)
    add_executable(nozzle_cuda src/cuda_solver.cu)
    set_target_properties(nozzle_cuda PROPERTIES
      CUDA_STANDARD 17
      CUDA_STANDARD_REQUIRED ON
      CUDA_ARCHITECTURES "${NOZZLE_CUDA_ARCHITECTURES}")
    target_compile_options(nozzle_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--expt-relaxed-constexpr>)
  else()
    message(STATUS "CUDA compiler not found: nozzle_cuda will not be built")
  endif()
endif()

if(NOZZLE_ENABLE_KOKKOS)
  if(NOZZLE_KOKKOS_ROOT)
    foreach(_tpl HWLOC MEMKIND LIBNUMA)
      if(_tpl STREQUAL "LIBNUMA")
        set(_lib_name numa)
      else()
        string(TOLOWER "${_tpl}" _lib_name)
      endif()
      find_library(_${_tpl}_LIBRARY NAMES ${_lib_name} HINTS "${NOZZLE_KOKKOS_ROOT}" PATH_SUFFIXES lib/x86_64-linux-gnu NO_DEFAULT_PATH REQUIRED)
      add_library(Kokkos::${_tpl} UNKNOWN IMPORTED)
      set_target_properties(Kokkos::${_tpl} PROPERTIES IMPORTED_LOCATION "${_${_tpl}_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${NOZZLE_KOKKOS_ROOT}/include")
    endforeach()
    list(PREPEND CMAKE_PREFIX_PATH "${NOZZLE_KOKKOS_ROOT}")
  endif()
  find_package(Kokkos 3.7 QUIET)
  if(Kokkos_FOUND)
    if(TARGET CUDA::cuda_driver)
      get_target_property(_cuda_driver_location CUDA::cuda_driver IMPORTED_LOCATION)
      if(NOT _cuda_driver_location)
        find_library(_cuda_driver_fallback NAMES libcuda.so.1 HINTS /usr/lib/x86_64-linux-gnu /usr/lib/wsl/lib NO_DEFAULT_PATH)
        if(_cuda_driver_fallback)
          set_target_properties(CUDA::cuda_driver PROPERTIES IMPORTED_LOCATION "${_cuda_driver_fallback}" IMPORTED_LOCATION_RELEASE "${_cuda_driver_fallback}")
        endif()
      endif()
    endif()
    add_executable(nozzle_kokkos src/kokkos_solver.cpp)
    target_link_libraries(nozzle_kokkos PRIVATE Kokkos::kokkos)
    target_compile_options(nozzle_kokkos PRIVATE -Wall -Wextra -Wpedantic)
  else()
    message(STATUS "Kokkos not found: run scripts/bootstrap_kokkos_ubuntu.sh and configure with -DNOZZLE_KOKKOS_ROOT=.deps/kokkos/usr")
  endif()
endif()

enable_testing()
add_test(NAME mapping_and_geometry COMMAND nozzle_solver --self-test)
add_test(NAME backend_geometry COMMAND backend_geometry_test)
add_test(NAME block_linear_solver COMMAND nozzle_solver --linear-test)
add_test(NAME muscl_numerics COMMAND muscl_numerics_test)
